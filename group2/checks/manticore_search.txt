FROM reviews
WHERE MATCH('отличный товар рекомендую')

2. Attribute filtering
================================================================================

SQL:
SELECT
    id,
    title,
    product_id,
    rating
FROM reviews
WHERE rating >= 4
  AND product_id = 42
ORDER BY
    rating DESC,
    id
LIMIT 20
